class Solution:
    def createGrid(self, m: int, n: int) -> list[str]:
        if m == 1:
            return ["." * n]

        if n == 1:
            return ["." for _ in range(m)]

        grid = [['#'] * n for _ in range(m)]

    # Open the first row, except the last column
        for j in range(n - 1):
            grid[0][j] = '.'

    # Create the path through the second row
        grid[1][n - 2] = '.'
        grid[1][n - 1] = '.'

    # Open the last column for the remaining rows
        for i in range(2, m):
            grid[i][n - 1] = '.'

        return [''.join(row) for row in grid]
